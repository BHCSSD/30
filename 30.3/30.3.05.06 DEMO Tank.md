## Day 3 
**Goal:** Switch from Mouse-follow to **Keyboard Controls** using Rotation and Trig
* make a copy in p5js for version control



### 1. Tank Class Update
```js
class Tank {
  constructor(x, y) {
    this.pos = createVector(x, y);
    this.bullets = [];
    this.angle = 0;      // NEW: Track facing direction
    this.speed = 3;      // NEW: Constant engine speed
    this.rotSpeed = 3;   // NEW: How fast we turn
  }

  movement() {
    // STEERING: Change angle based on Left/Right arrows
    if (keyIsDown(LEFT_ARROW)) {
      this.angle -= this.rotSpeed;
    } // end if
    if (keyIsDown(RIGHT_ARROW)) {
      this.angle += this.rotSpeed;
    } // end if

    // ENGINE: Use Trig to move forward at the current angle
    if (keyIsDown(UP_ARROW)) {
      let vx = cos(this.angle) * this.speed; // X-velocity
      let vy = sin(this.angle) * this.speed; // Y-velocity

      this.pos.x += vx;
      this.pos.y += vy;
    } // end if
  } // end movement

  shoot() {
    // CHANGE: Give bullet the current tank angle
    let b = new Bullet(this.pos.x, this.pos.y, this.angle);
    this.bullets.push(b);
  } // end shoot

  update() {
    this.movement(); // NEW: Check keys instead of lerp

    for (let i = this.bullets.length - 1; i >= 0; i--) {
      this.bullets[i].move();
      if (this.bullets[i].isOffScreen()) {
        this.bullets.splice(i, 1);
      } // end if
    } // end for
  } // end update

  display() {
    push();
      translate(this.pos.x, this.pos.y); // NEW: Move origin to tank
      rotate(this.angle);                // NEW: Rotate the canvas
      
      fill('gray');
      rectMode(CENTER);
      rect(0, 0, 100, 60, 5);       
      rect(40, 0, 80, 10, 3);       
      rect(0, 0, 40, 40, 3);        
    pop();
    
    for (let b of this.bullets) {
      b.display();
    } // end for
  } // end display
} // end Tank class
```

### 2. Bullet Class Update
```js
class Bullet {
  constructor(x, y, angle) {
    this.pos = createVector(x, y);
    this.angle = angle; // NEW: Bullet "remembers" fire direction
    this.speed = 10;
    this.w = 40;
    this.h = 20;
  }

  move() {
    // NEW: Use Trig to fly in the direction it was shot
    this.pos.x += cos(this.angle) * this.speed;
    this.pos.y += sin(this.angle) * this.speed;
  } // end move

  display() {
    push();
      translate(this.pos.x, this.pos.y);
      rotate(this.angle); // NEW: Bullet faces its flight path
      rectMode(CENTER);
      fill("black");
      noStroke();
      rect(0, 0, this.w, this.h, 0, 15, 15, 0);
    pop();
  } // end display

  isOffScreen() {
    // NEW: Check all 4 walls since bullet can fly any direction
    return (this.pos.x > width || this.pos.x < 0 || 
            this.pos.y > height || this.pos.y < 0);
  } // end isOffScreen
} // end Bullet class
```
