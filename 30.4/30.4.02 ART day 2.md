function setup() {
  createCanvas(500, 500)
  colorMode(HSB, 360, 100, 100)
  noStroke()
}

function draw() {
  background(20);
  
  //Start the recursion in the center of the canvas:
  drawCircle(width / 2, height / 2, 450, 0);
}

function drawCircle(x, y, d, hueValue) {
  //1. BASE CASE: Stop when the circle gets smaller than 15 pixels
  if (d < 15) {
    return
  }

  // 2.DO THE WORK: Set the color and draw the current circle
  fill(hueValue % 360, 80, 95)
  circle(x, y, d)

  //3. RECURSIVE STEP: Call the exact same function, 
  // but make the circle 30px smaller and shift the color by 25
  drawCircle(x, y, d - 30, hueValue + 25);
}
