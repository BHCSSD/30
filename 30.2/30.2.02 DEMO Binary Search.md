def binarySearch_assert():
    arr = [1, 3, 5, 7, 9, 11, 13, 15]

    # Standard cases
    assert binarySearch(arr, 7) == 3
    assert binarySearch(arr, 1) == 0      # First element
    assert binarySearch(arr, 15) == 7     # Last element
    
    # Missing cases
    assert binarySearch(arr, 4) == -1
    assert binarySearch(arr, -10) == -1   # Below minimum
    assert binarySearch(arr, 100) == -1  # Above maximum
    
    # Edge cases (list size)
    assert binarySearch([], 5) == -1        # Empty list
    assert binarySearch([10], 10) == 0     # One element (found)
